cat >> ~/.bashrc <<'EOF'

\# --- auto history logging ---

export HISTFILE=~/.bash\_history

export HISTSIZE=100000

export HISTFILESIZE=200000

shopt -s histappend

export PROMPT\_COMMAND='history -a'

EOF



source ~/.bashrc

-------------------------------------

history > history.txt



---------------------------명령어 히스토리(history) 를 “오늘 날짜 파일명”으로 저장

history > "history\_$(date +%Y-%m-%d).txt"



---------------------------------------특정 범위(예: 최근 200개만):

history 200 > "history\_$(date +%Y-%m-%d)\_last200.txt"



------------------------------------앞으로부터 시간까지 기록되게 설정(추천)

echo 'export HISTTIMEFORMAT="%F %T  "' >> ~/.bashrc

source ~/.bashrc



-------------------------------------2) 오늘 날짜만 필터해서 저장

history | grep "^$(date +%F)" > "history\_$(date +%F)\_today.txt"



-----------------------C) 로그 파일 자체를 “날짜별”로 자동 분리(출력까지 남길 때)

git status 2>\&1 | tee -a "session\_$(date +%F).log"
